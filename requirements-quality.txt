# Optional: the GPU voice. Installed when the image is built with
# TTS_PROFILE=quality.
#
# Chatterbox pins torch exactly (torch==2.6.0 as of 0.1.7), so it must bring its
# own torch instead of being layered on top of one we chose. That is why nothing
# installs torch before this file -- letting pip resolve it keeps the pins
# consistent, at the cost of not picking the CUDA build ourselves. Torch ships its
# CUDA libraries as wheels, so it works on any recent driver.
#
# NOT included: qwen3-tts. The PyPI package requires Python >= 3.13 and the CUDA
# base images ship 3.10/3.12. See README ("Bessere Stimme") for how to run it in
# its own container and reach it over HTTP via the openai TTS backend.

chatterbox-tts>=0.1.7

# Reading reference audio for voice cloning
soundfile>=0.12

# CPU embeddings, so the knowledge base costs no VRAM
fastembed>=0.4

# Piper on the GPU. Synthesis is the largest share of the response time, and this
# takes its realtime factor from roughly 0.3 to under 0.05 for a few hundred MB of
# VRAM. Enable with tts.piper.use_cuda: true once installed.
# Replaces the CPU build of onnxruntime, so it belongs in this profile only.
onnxruntime-gpu>=1.19
